# 取圖第三步改成欄位按鈕＋選擇器 實作計畫

> **給代理執行者：** 必用子技能：superpowers:subagent-driven-development（建議）或 superpowers:executing-plans，逐任務執行。步驟用 `- [ ]` 勾選追蹤。
> 回報與報告一律用繁體中文寫。

**目標：** 把取圖精靈第三步（與共用同一個畫面的批次編輯）從「快捷列＋詳細資訊抽屜＋【套用】」改成「底部一列欄位按鈕＋各欄位自己的選擇器，按【確定】即寫入勾選的那幾張」，縮圖左下顯示四個欄位圖示，長按看單張圖資。

**架構：** 狀態機 `Step3Store` 換成單一入口 `applyToSelected(DetailsPatch)`，另外算出「某欄位是否每一張都有值」與「能不能完成」。畫面 `Step3DetailsScreen` 只剩縮圖牆＋底部列，四個選擇器（月曆對話框、地點／標籤／描述抽屜）各自決定「有沒有動過」，動過才往上送 `DetailsPatch`。精靈與批次編輯的 ViewModel 都改成收 `DetailsPatch`。

**技術：** Kotlin、Jetpack Compose（Material 3 `ModalBottomSheet`、`DatePickerDialog`、`Popup`）、JUnit、Robolectric Compose 測試。

**規格：** [`docs/superpowers/specs/2026-09-10-videoshot-app-design.md`](../specs/2026-09-10-videoshot-app-design.md) 第五節「第三步：定義圖資」與「完成」；驗收標準是 [`mockups/uiux-v2/驗收操作手冊.md`](../../../mockups/uiux-v2/驗收操作手冊.md) §四第三步。

enum class DetailField { DATE, PLACE, TAGS, DESCRIPTION }

/** 這一張的這個欄位有沒有值。縮圖左下的圖示亮不亮、欄位按鈕打不打勾都看這個。 */
fun ShotDetails.has(field: DetailField): Boolean = when (field) {
    DetailField.DATE -> eventDate.isNotBlank()
    DetailField.PLACE -> !place.isNullOrBlank()
    DetailField.TAGS -> tags.isNotEmpty()
    DetailField.DESCRIPTION -> !description.isNullOrBlank()
}

    /**
     * 每一張都有這個欄位的值（規格第五節第三步的欄位表「打勾條件」）。

在 `Step3Store` 裡（`applyPatch()` 之前）加：

```kotlin
    /**
     * 選擇器按【確定】：把 [patch] 寫進目前勾選的每一張（規格第五節「寫入語意」）。
     * 「沒動過就不改」由選擇器自己判斷，沒動過就不會呼叫到這裡；這裡只負責沒動過的**欄位**原封不動。
     * 勾選不變 —— 使用者常常接著替同一批設下一個欄位。
     */
    fun applyToSelected(patch: DetailsPatch) {
        val current = _state.value
        if (current.selected.isEmpty() || patch.isEmpty) return
        // 「新加」要用套用前的圖資判斷：標籤選擇器送出的是整份清單（起始值是勾選格子的共同標籤）
        val before = current.selected.mapNotNull { current.details[it] }
        val beforeTags = before.flatMapTo(mutableSetOf()) { it.tags }
        val newPlace = patch.place?.trim()?.takeIf { it.isNotEmpty() && before.any { d -> d.place != it } }
        _state.value = current.copy(
            details = applyToCells(current.details, current.selected, patch),
            usedPlaces = current.usedPlaces + listOfNotNull(newPlace),
            usedTags = current.usedTags + (normalizeTags(patch.tags.orEmpty()) - beforeTags),
        )
    }
```

@OptIn(ExperimentalLayoutApi::class)
@Composable
internal fun PlacePickerSheet(
    field: Common<String?>,
    count: Int,
    suggestions: List<String>,
    onConfirm: (DetailsPatch) -> Unit,
    onDismiss: () -> Unit,
) {
    val initial = (field as? Common.One<String?>)?.value.orEmpty()
    var input by remember { mutableStateOf(initial) }
    val query = input.trim()
    PickerSheet(
        title = "地點",
        count = count,
        canClear = field.hasAnyValue(),
        canConfirm = query.isNotEmpty(),
        onClear = { onConfirm(DetailsPatch(place = "")) },
        onConfirm = { if (query == initial) onDismiss() else onConfirm(DetailsPatch(place = query)) },
        onDismiss = onDismiss,
    ) {

            value = input,
            onValueChange = { input = it },
            placeholder = if (field is Common.Mixed) MIXED else "輸入或從下面挑",
            mixedPlaceholder = field is Common.Mixed,
            semanticLabel = "地點名稱",

        val places = remember(suggestions, query) {
            val filtering = query.isNotEmpty() && query !in suggestions
            suggestions.filter { it != query && (!filtering || it.contains(query, ignoreCase = true)) }

        VsChipPager(names = places, kindOf = { ChipKind.PLACE }, onClick = { i -> input = places[i] }, resetKey = query)

 * 標籤選擇器。由上到下：已選的標籤（可點 ✕ 移除）→ 輸入框＋【加入】（鍵盤 Enter 同【加入】）→ 用過的標籤。
 * 【確定】＝**整組取代**勾選那幾張的標籤；全部移除再確定＝清除。新打的標籤當成「其他」（不另外問種類）。
 * 多個值時起始是空的、顯示〈多個值〉；沒加任何標籤就按確定＝沒動過。
 */
@OptIn(ExperimentalLayoutApi::class)
@Composable
internal fun TagPickerSheet(
    field: Common<List<String>>,
    count: Int,
    suggestions: List<TagNameKind>,
    onConfirm: (DetailsPatch) -> Unit,
    onDismiss: () -> Unit,
) {
    val initial = (field as? Common.One<List<String>>)?.value.orEmpty()
    var tags by remember { mutableStateOf(initial) }
    var typing by remember { mutableStateOf("") }
    val kinds = remember(suggestions) { suggestions.associate { it.name to it.kind } }
    fun kindOf(tag: String) = ChipKind.ofTagKind(kinds[tag] ?: "other")
    fun add(raw: String) {
        normalizeTags(listOf(raw)).firstOrNull()?.let { tags = normalizeTags(tags + it) }
        typing = ""
    }
    PickerSheet(
        title = "標籤",
        count = count,
        canClear = field.hasAnyTag(),
        canConfirm = true,
        onClear = { onConfirm(DetailsPatch(tags = emptyList())) },
        onConfirm = {
            // 「動過」是拿現值跟起始值比：加了又拿掉＝沒動過
            if (normalizeTags(tags) == normalizeTags(initial)) onDismiss() else onConfirm(DetailsPatch(tags = normalizeTags(tags)))
        },
        onDismiss = onDismiss,
    ) {
        if (tags.isNotEmpty()) {
            FlowRow(horizontalArrangement = Arrangement.spacedBy(AppTheme.spacing.s2)) {
                tags.forEach { tag -> VsRemovableTag(tag, kindOf(tag), onRemove = { tags = tags - tag }) }
            }
        } else if (field is Common.Mixed) {
            Text(MIXED, style = MaterialTheme.typography.bodyMedium.copy(fontStyle = FontStyle.Italic), color = AppTheme.colors.textDim)
        }
        Row(horizontalArrangement = Arrangement.spacedBy(AppTheme.spacing.s2), verticalAlignment = Alignment.Bottom) {
            VsTextField(
                value = typing,
                onValueChange = { typing = it },
                modifier = Modifier.weight(1f),
                placeholder = "新增標籤",
                semanticLabel = "新增標籤",
                keyboardOptions = KeyboardOptions(imeAction = ImeAction.Done),
                keyboardActions = KeyboardActions(onDone = { add(typing) }),
            )
            VsButton("加入", onClick = { add(typing) }, variant = ButtonVariant.Secondary)
        }
        val q = typing.trim()
        val options = remember(suggestions, tags, q) {
            suggestions.filter { it.name !in tags && (q.isEmpty() || it.name.contains(q, ignoreCase = true)) }
        }
        VsChipPager(
            names = remember(options) { options.map { it.name } },
            kindOf = { i -> ChipKind.ofTagKind(options[i].kind) },
            onClick = { i -> add(options[i].name) },
            resetKey = q,
        )
    }
}

/**
 * 描述選擇器：多行輸入框。改過才送（清空也算改過＝清除）；多個值時起始是空的、顯示〈多個值〉，沒打字就按確定＝沒動過。
 */
@Composable
internal fun DescriptionPickerSheet(
    field: Common<String?>,
    count: Int,
    onConfirm: (DetailsPatch) -> Unit,
    onDismiss: () -> Unit,
) {
    val initial = (field as? Common.One<String?>)?.value.orEmpty()
    var input by remember { mutableStateOf(initial) }
    PickerSheet(
        title = "描述",
        count = count,
        canClear = field.hasAnyValue(),
        canConfirm = true,
        onClear = { onConfirm(DetailsPatch(description = "")) },
        onConfirm = { if (input == initial) onDismiss() else onConfirm(DetailsPatch(description = input)) },
        onDismiss = onDismiss,
    ) {
        VsTextField(
            value = input,
            onValueChange = { input = it },
            placeholder = if (field is Common.Mixed) MIXED else null,
            mixedPlaceholder = field is Common.Mixed,
            semanticLabel = "描述",
            lines = 3,
        )
    }
}

/**

 * 鍵盤開著時整塊被頂上去（`imePadding`），按鈕不會被蓋住。
 */
@Composable
private fun PickerSheet(
    title: String,
    count: Int,
    canClear: Boolean,
    canConfirm: Boolean,
    onClear: () -> Unit,
    onConfirm: () -> Unit,
    onDismiss: () -> Unit,
    content: @Composable ColumnScope.() -> Unit,

                .fillMaxWidth()
                .imePadding()
                .padding(horizontal = AppTheme.spacing.s4)
                .padding(top = AppTheme.spacing.s3, bottom = AppTheme.spacing.s5),
            verticalArrangement = Arrangement.spacedBy(AppTheme.spacing.s3),
        ) {

                VsButton("確定", onClick = onConfirm, modifier = Modifier.weight(2f), enabled = canConfirm)
            }
        }
    }
}

/**
 * 第三步：定義圖資（2026-10-18 scott 改版：原本的快捷列＋詳細資訊抽屜＋【套用】太難操作）。
 *
 * 畫面只有縮圖牆與底部一列：【全選】【時間】【地點】【標籤】【描述】＋【完成】。
 * 四個欄位按鈕各開自己的選擇器，**按【確定】的那一刻就寫進目前勾選的那幾張**（[onApply]），沒有另外的【套用】；
 * 「沒動過就不改」由選擇器自己判斷（見 `Step3Pickers.kt`）。
 *
 * 縮圖牆跟第二步一樣：**每列 4 張正方形、左右貼齊螢幕、格線 [ThumbGridGap]**。
 *
 * @param requireTimeAndPlace 取圖精靈是 true：每一張都有時間與地點（[Step3State.canFinish]）才完成；
 *        還不能完成時【完成】是淡化的，按了跳提示。批次編輯是 false —— 只寫回改過的那幾張，隨時可完成。
 */
@Composable
fun Step3DetailsScreen(
    state: Step3State,
    bitmapFor: suspend (Int) -> ImageBitmap?,
    onToggle: (Int) -> Unit,
    onSelectAll: () -> Unit,
    onApply: (DetailsPatch) -> Unit,
    onFinish: () -> Unit,
    modifier: Modifier = Modifier,
    placeSuggestions: List<String> = emptyList(),
    tagSuggestions: List<TagNameKind> = emptyList(),
    requireTimeAndPlace: Boolean = false,
) {
    val snackbar = remember { SnackbarHostState() }
    val scope = rememberCoroutineScope()
    // 新的提示蓋掉舊的：連按兩次不會排隊等兩輪
    fun hint(text: String) {
        scope.launch {
            snackbar.currentSnackbarData?.dismiss()
            snackbar.showSnackbar(text, duration = SnackbarDuration.Short)
        }
    }
    var picking by remember { mutableStateOf<DetailField?>(null) }
    fun open(field: DetailField) {
        if (state.selected.isEmpty()) hint("先勾選圖片") else picking = field
    }
    val canFinish = !requireTimeAndPlace || state.canFinish

    Box(modifier.fillMaxSize()) {
        Column(Modifier.fillMaxSize()) {
            // sheet 已經在本機，這一行幾乎一閃而過（規格第五節）
            state.progressText?.let {
                Text(
                    it,
                    style = MaterialTheme.typography.bodySmall,
                    color = AppTheme.colors.textDim,
                    modifier = Modifier.padding(horizontal = 16.dp, vertical = 4.dp),
                )
            }

            LazyVerticalGrid(
                columns = GridCells.Fixed(4),
                modifier = Modifier.weight(1f).fillMaxWidth(),
                horizontalArrangement = Arrangement.spacedBy(ThumbGridGap),
                verticalArrangement = Arrangement.spacedBy(ThumbGridGap),
            ) {
                items(state.cells, key = { it.cell }) { cell ->
                    Step3Thumb(
                        cell = cell,
                        details = state.details[cell.cell],
                        selected = cell.cell in state.selected,
                        bitmapFor = bitmapFor,
                        onToggle = { onToggle(cell.cell) },
                    )
                }
            }

            VsActionDock {
                // 五顆欄位按鈕彼此不留間距（各自 48dp 寬、已有留白），【完成】才拿得到夠寬的位置
                Row {
                    VsFieldButton(VsIcons.ImagesCheck, "全選", onSelectAll)
                    DetailField.entries.forEach { f ->
                        VsFieldButton(f.icon, f.label, onClick = { open(f) }, done = state.isComplete(f))
                    }
                }
                VsButton(
                    "完成",
                    onClick = { if (canFinish) onFinish() else hint("每一張都要先設定時間和地點") },
                    modifier = Modifier
                        .weight(1f)
                        .semantics { if (!canFinish) stateDescription = "還不能完成" },
                    dimmed = !canFinish,
                )
            }
        }
        // 浮在底部列正上方（同精靈外殼的「已截圖」提示）
        SnackbarHost(
            snackbar,
            Modifier.align(Alignment.BottomCenter).navigationBarsPadding().padding(bottom = AppTheme.spacing.navHeight),
        )
    }

    val count = state.selected.size
    val close = { picking = null }
    val confirm: (DetailsPatch) -> Unit = { onApply(it); picking = null }
    when (picking) {
        DetailField.DATE -> {
            val one = (state.eventDateField as? Common.One<String>)?.value
            // 選回勾選那幾張原本一致的那天＝沒動過
            VsDatePickerDialog(initial = one, onPick = { d -> if (d != one) onApply(DetailsPatch(eventDate = d)) }, onDismiss = close)
        }
        DetailField.PLACE -> PlacePickerSheet(state.placeField, count, placeSuggestions, confirm, close)
        DetailField.TAGS -> TagPickerSheet(state.tagsField, count, tagSuggestions, confirm, close)
        DetailField.DESCRIPTION -> DescriptionPickerSheet(state.descriptionField, count, confirm, close)
        null -> Unit
    }
}

/**
 * 一格（正方形、直角，同第二步）。左上打勾徽章＝勾選中；左下四個小圖示＝時間／地點／標籤／描述，有值亮白、沒有半透明。
 * （2026-10-18 拿掉右上的已套用綠勾與右下的時間：這一步的重點是替圖填資料，時間點在長按浮窗裡看。）
 *
 * 顏色不是唯一的訊號 —— 勾選與各欄位有沒有值都寫進整格的 `contentDescription`（手冊 §零）；
 * 徽章與小圖示本身是純裝飾，不另設名稱。
 */
@Composable
private fun Step3Thumb(
    cell: Step3Cell,
    details: ShotDetails?,
    selected: Boolean,
    bitmapFor: suspend (Int) -> ImageBitmap?,
    onToggle: () -> Unit,
) {
    val label = thumbLabel(cell, details, selected)
    val cellShape = RectangleShape
    Box(
        Modifier
            .aspectRatio(1f)
            .clip(cellShape)
            .background(AppTheme.colors.surface2)
            .focusRing(cellShape)
            .clickable { onToggle() }
            .semantics { contentDescription = label },
    ) {
        val bitmap by produceState<ImageBitmap?>(null, cell.cell, bitmapFor) { value = bitmapFor(cell.cell) }
        bitmap?.let {
            Image(bitmap = it, contentDescription = null, contentScale = ContentScale.Crop, modifier = Modifier.fillMaxSize())
        }
        if (selected) VsSelectedBadge(Modifier.align(Alignment.TopStart), semanticLabel = null)
        Row(
            Modifier
                .align(Alignment.BottomStart)
                .padding(AppTheme.spacing.s1)
                .clip(RoundedCornerShape(AppTheme.radii.sm))
                .background(AppTheme.colors.scrim)
                .padding(horizontal = 3.dp, vertical = 2.dp),
            horizontalArrangement = Arrangement.spacedBy(2.dp),
        ) {
            DetailField.entries.forEach { f ->
                val on = details?.has(f) == true
                Icon(f.icon, contentDescription = null, tint = Color.White.copy(alpha = if (on) 1f else 0.35f), modifier = Modifier.size(12.dp))
            }
        }
    }
}

/** 「第三步的格子 01:10・已勾選・已設定：時間、地點・未設定：標籤、描述」。 */
internal fun thumbLabel(cell: Step3Cell, details: ShotDetails?, selected: Boolean): String = buildString {
    append("第三步的格子 ")
    append(formatClock(cell.atSec))
    if (selected) append("・已勾選")
    val (set, unset) = DetailField.entries.partition { details?.has(it) == true }
    if (set.isNotEmpty()) append("・已設定：").append(set.joinToString("、") { it.label })
    if (unset.isNotEmpty()) append("・未設定：").append(unset.joinToString("、") { it.label })
}

@Composable
internal fun Step3InfoPopup(cell: Step3Cell, details: ShotDetails?, onDismiss: () -> Unit) {
    val gap = with(LocalDensity.current) { 6.dp.roundToPx() }
    val position = remember(gap) { BelowOrAbove(gap) }
    Popup(popupPositionProvider = position, onDismissRequest = onDismiss, properties = PopupProperties(focusable = true)) {
        Column(
            Modifier
                .widthIn(max = 260.dp)
                .background(Color.Black.copy(alpha = 0.78f), RoundedCornerShape(10.dp))
                .clickable(onClick = onDismiss)
                .padding(horizontal = 12.dp, vertical = 10.dp),
            verticalArrangement = Arrangement.spacedBy(4.dp),
        ) {
            val unset = "未設定"
            InfoRow(DetailField.DATE, details?.eventDate?.takeIf { it.isNotBlank() }?.let(::chineseDate) ?: unset)
            InfoRow(DetailField.PLACE, details?.place?.takeIf { it.isNotBlank() } ?: unset)
            InfoRow(DetailField.TAGS, details?.tags?.takeIf { it.isNotEmpty() }?.joinToString("、") ?: unset)
            InfoRow(DetailField.DESCRIPTION, details?.description?.takeIf { it.isNotBlank() } ?: unset)
            Text(
                "影片 ${formatClock(cell.atSec)}",
                style = MaterialTheme.typography.labelSmall,
                color = Color.White.copy(alpha = 0.7f),
            )
        }
    }
}

@Composable
private fun InfoRow(field: DetailField, value: String) {
    Row(horizontalArrangement = Arrangement.spacedBy(6.dp), verticalAlignment = Alignment.Top) {
        Icon(field.icon, contentDescription = field.label, tint = Color.White, modifier = Modifier.size(16.dp))
        Text(value, style = MaterialTheme.typography.bodyMedium, color = Color.White)
    }
}

/** 放在格子下方；下方放不下就放上方。左右夾在視窗內（最左、最右那欄的格子浮窗不會超出螢幕）。 */
private class BelowOrAbove(private val gapPx: Int) : PopupPositionProvider {
    override fun calculatePosition(
        anchorBounds: IntRect,
        windowSize: IntSize,
        layoutDirection: LayoutDirection,
        popupContentSize: IntSize,
    ): IntOffset {
        val x = (anchorBounds.left + (anchorBounds.width - popupContentSize.width) / 2)
            .coerceIn(0, (windowSize.width - popupContentSize.width).coerceAtLeast(0))
        val below = anchorBounds.bottom + gapPx
        val y = if (below + popupContentSize.height <= windowSize.height) below
        else (anchorBounds.top - gapPx - popupContentSize.height).coerceAtLeast(0)
        return IntOffset(x, y)
    }
}
